# Function to fit model comparing left to right colon and extract stats :
fit_location_model <- function(sig, data) {
  model <- lmer(count ~ site + (1 | patient), data = filter(data, signature == sig))
  tidy(model, effects = "fixed") %>%
    filter(term != "(Intercept)") %>%   # site comparison term
    mutate(
      signature = sig, estimate = round(estimate, 0), 
      low_ci = round(estimate - 1.96 * std.error, 0), 
      upp_ci = round(estimate + 1.96 * std.error, 0), p.value = round(p.value, 10)) %>%
    dplyr::select(signature, estimate, low_ci, upp_ci, p.value)
}
